cmake_minimum_required(VERSION 3.25)
project(rocell_windows_usb_identity VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Requires existing Windows SDK/MSVC")
endif()
add_library(usb_identity_core STATIC admission.cpp observation.cpp serialize.cpp)
target_link_libraries(usb_identity_core PUBLIC bcrypt)
add_executable(rocell_windows_usb_identity main.cpp windows_api.cpp admission_entry.cpp)
target_link_libraries(rocell_windows_usb_identity PRIVATE usb_identity_core setupapi cfgmgr32)
add_executable(rocell_usb_identity_tests tests.cpp)
target_link_libraries(rocell_usb_identity_tests PRIVATE usb_identity_core)
# This child is separately linked: it has NO WindowsUsbIdentityApi/device code.
add_executable(rocell_usb_identity_entry_tests entry_tests.cpp admission_entry.cpp)
target_link_libraries(rocell_usb_identity_entry_tests PRIVATE usb_identity_core)
foreach(target usb_identity_core rocell_windows_usb_identity rocell_usb_identity_tests rocell_usb_identity_entry_tests)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
endforeach()
foreach(target rocell_windows_usb_identity rocell_usb_identity_tests rocell_usb_identity_entry_tests)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
endforeach()
enable_testing()
add_test(NAME usb_identity_incapable COMMAND rocell_usb_identity_tests)
set_tests_properties(usb_identity_incapable PROPERTIES TIMEOUT 15)
get_filename_component(ROCELL_TEST_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/../../../.venv/Scripts/python.exe" ABSOLUTE)
if(EXISTS "${ROCELL_TEST_PYTHON}")
  add_test(NAME usb_identity_wire_incapable COMMAND "${ROCELL_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/wire_test.py" "$<TARGET_FILE:rocell_usb_identity_tests>")
  add_test(NAME usb_identity_admission_incapable COMMAND "${ROCELL_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/entry_test.py" "$<TARGET_FILE:rocell_usb_identity_entry_tests>" "$<TARGET_FILE:rocell_usb_identity_tests>")
  set_tests_properties(usb_identity_wire_incapable usb_identity_admission_incapable PROPERTIES TIMEOUT 20)
endif()
