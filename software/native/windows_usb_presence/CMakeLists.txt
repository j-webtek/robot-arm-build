cmake_minimum_required(VERSION 3.25)
project(rocell_windows_usb_presence VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Requires existing Windows SDK/MSVC")
endif()
# Reuse only the strict flat JSON parser and SHA implementation. This does not
# rebuild, repin or modify the prior descriptor-query executable.
add_library(usb_presence_core STATIC presence.cpp ../windows_usb_identity/admission.cpp)
target_link_libraries(usb_presence_core PUBLIC bcrypt)
add_executable(rocell_windows_usb_presence main.cpp entry.cpp windows_api.cpp)
target_link_libraries(rocell_windows_usb_presence PRIVATE usb_presence_core cfgmgr32)
# No WindowsPresenceApi or cfgmgr32 dependency in either incapable test binary.
add_executable(rocell_usb_presence_tests tests.cpp)
target_link_libraries(rocell_usb_presence_tests PRIVATE usb_presence_core)
add_executable(rocell_usb_presence_entry_tests fixture_main.cpp entry.cpp)
target_link_libraries(rocell_usb_presence_entry_tests PRIVATE usb_presence_core)
foreach(target usb_presence_core rocell_windows_usb_presence rocell_usb_presence_tests rocell_usb_presence_entry_tests)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
endforeach()
foreach(target rocell_windows_usb_presence rocell_usb_presence_tests rocell_usb_presence_entry_tests)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
endforeach()
enable_testing()
add_test(NAME usb_presence_incapable COMMAND rocell_usb_presence_tests)
set_tests_properties(usb_presence_incapable PROPERTIES TIMEOUT 15)
get_filename_component(ROCELL_TEST_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/../../../.venv/Scripts/python.exe" ABSOLUTE)
if(EXISTS "${ROCELL_TEST_PYTHON}")
  add_test(NAME usb_presence_wire_incapable COMMAND "${ROCELL_TEST_PYTHON}" -m pytest "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/unit/test_usb_presence_protocol.py" -q)
  add_test(NAME usb_presence_admission_incapable COMMAND "${ROCELL_TEST_PYTHON}" -m pytest "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/unit/test_usb_presence_entry.py" -q)
  set_tests_properties(usb_presence_wire_incapable usb_presence_admission_incapable PROPERTIES TIMEOUT 20)
endif()
